# the mmap after loading (see _materialize_cpu_weights), so no pin is required.

torch
# >= 4: the worker builds the cross-encoder with the `activation_fn` / `model_kwargs`
# constructor so it always scores raw logits (see _load_reranker).
sentence-transformers>=4
